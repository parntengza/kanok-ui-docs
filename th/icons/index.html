ติดตั้ง

Carbon แยกไอคอนเป็นสองแพ็กเกจ ทั้งคู่ใช้ license Apache-2.0 ให้ติดตั้งตัวที่ตรงกับ stack ของคุณ @purpurabog/ui ไม่ได้พึ่งแพ็กเกจไหนในสองตัวนี้

ReactHTML, Lit และ framework อื่นnpm install @carbon/icons-react npm install @carbon/icons @carbon/icon-helpers

หาชื่อไอคอน

ใน library ชื่อไอคอนเป็นตัวพิมพ์เล็กคั่นด้วยขีด เช่น trash-can แต่ละแพ็กเกจเขียนชื่อต่างกันแบบนี้

ใน library React (@carbon/icons-react) อื่น ๆ (@carbon/icons)
add Add @carbon/icons/es/add/16.js
trash-can TrashCan @carbon/icons/es/trash-can/16.js
overflow-menu--horizontal OverflowMenuHorizontal @carbon/icons/es/overflow-menu--horizontal/16.js
chevron--right ChevronRight @carbon/icons/es/chevron--right/16.js

@carbon/icons มีไฟล์แยกตามขนาด คือ 16.js, 20.js, 24.js และ 32.js บางไอคอนวาดต่างกันที่ขนาด 16px จึงควรเลือกขนาดที่ component แสดง (ดู ขนาดและสี)

ใส่ไอคอนใน component

ใน React ไอคอนคือ component ที่ render <svg> และส่ง prop อื่น เช่น slot ต่อให้ <svg> ถ้าไม่มี bundler หรือใช้ framework อื่น ให้แปลงข้อมูลไอคอนเป็น <svg> ด้วย toString จาก @carbon/icon-helpers ส่วนหน้า HTML ธรรมดา ให้กด Download SVG ใน library แล้วแปะ <svg> ลงไปตรง ๆ แบบที่ตัวอย่าง HTML ในเว็บนี้ทำ

LitJavaScriptReactReact + Tailwindimport { html } from 'lit'; import { unsafeHTML } from 'lit/directives/unsafe-html.js'; import { toString } from '@carbon/icon-helpers'; import Add16 from '@carbon/icons/es/add/16.js'; import TrashCan16 from '@carbon/icons/es/trash-can/16.js'; /** ไอคอนของ Carbon เป็น markup พร้อม attribute เพิ่ม เช่น `slot` */ const icon = (descriptor, attrs = {}) => unsafeHTML(toString({ ...descriptor, attrs: { ...descriptor.attrs, ...attrs } })); html` <ui-button kind="primary">เพิ่ม ${icon(Add16, { slot: 'suffix' })}</ui-button> <ui-icon-button label="ลบ">${icon(TrashCan16)}</ui-icon-button> `; import { toString } from '@carbon/icon-helpers'; import Add16 from '@carbon/icons/es/add/16.js'; const button = document.querySelector('ui-button'); button.insertAdjacentHTML('beforeend', toString({ ...Add16, attrs: { ...Add16.attrs, slot: 'suffix' } })); import { UiMenuItem, UiOverflowMenu } from '@purpurabog/ui/react'; import { Copy, Edit } from '@carbon/icons-react'; <UiOverflowMenu label="ตัวเลือกเอกสาร"> <UiMenuItem label="แก้ไข"><Edit slot="icon" /></UiMenuItem> <UiMenuItem label="ทำสำเนา"><Copy slot="icon" /></UiMenuItem> </UiOverflowMenu> import { MenuItem, OverflowMenu } from '@/components/ui/react-tw/menu'; import { Copy, Edit } from '@carbon/icons-react'; <OverflowMenu label="ตัวเลือกเอกสาร"> <MenuItem label="แก้ไข" icon={<Edit />} /> <MenuItem label="ทำสำเนา" icon={<Copy />} /> </OverflowMenu>

ขนาดและสี

component เป็นตัวกำหนดขนาดไอคอน: 16px ในปุ่ม เมนู tag และ side nav, 20px ใน header และแท็บขนาดใหญ่ที่มีแต่ไอคอน, 32px หน้าชื่อของ page header ดังนั้น width และ height ที่ติดมากับไอคอนจึงไม่มีผลในที่เหล่านี้ เมื่อ component แสดงไอคอนที่ 20px ขึ้นไป ให้ใช้ไอคอนขนาดนั้น (size={20} ใน React, 20.js ใน @carbon/icons) จะได้ลายเส้นที่วาดมาสำหรับขนาดนั้น

ไอคอนของ Carbon ลงสีด้วย currentColor จึงใช้สีตัวอักษรของ component ทั้งตอน hover ตอน disabled และในทุกธีม ไม่ต้องตั้ง fill เอง

การเข้าถึง

ไอคอนที่อยู่ใน control เป็นแค่ของตกแต่ง ทั้งสองแพ็กเกจใส่ aria-hidden="true" ให้ไอคอนที่ไม่มี aria-label เอง ชื่อของ control มาจากตัว control เช่น label ของปุ่มไอคอน ซึ่งเป็นข้อความ tooltip ด้วย อย่าใส่ชื่อให้ไอคอนซ้ำอีก ไม่อย่างนั้น screen reader จะอ่านชื่อสองรอบ

ไอคอนที่อยู่เดี่ยว ๆ และมีความหมาย (เช่น บอกสถานะ) ต้องมีชื่อของตัวเอง ให้ส่ง aria-label (React) หรือใส่ไว้ใน attrs แล้วไอคอนจะได้ role="img" ถ้าเป็นสถานะ status indicator มีไอคอนและชื่อที่ถูกต้องให้อยู่แล้ว

ชื่อที่ซ้ำกัน

ไอคอนบางตัวชื่อซ้ำกับ component เช่น Search, Notification หรือ Copy ให้เปลี่ยนชื่อตอน import ตัวอย่างในเว็บนี้เติม Icon ต่อท้าย

import { Search as SearchIcon, Notification as NotificationIcon } from '@carbon/icons-react';
