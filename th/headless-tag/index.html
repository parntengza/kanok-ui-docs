Headless Tag

Tag คือพฤติกรรมของ Tag แบบมี style แยกเป็น React part ไม่มี style สามชิ้น Tag.Root คือ tag แบบอ่านอย่างเดียวหรือแบบปิดได้ (เป็น <span>) Tag.Dismiss คือปุ่มปิดของมัน กดแล้วเรียก onDismiss และ click handler ที่ครอบ tag อยู่จะไม่เห็นการกดนี้ Tag.Toggle คือ tag แบบเปิดปิดได้ เป็น <button aria-pressed> ที่สลับเมื่อกด กด Space หรือ Enter ส่วน tag แบบกดเพื่อเปิดเป็น <button> ธรรมดา จึงไม่ต้องมี part

markup และ style ทั้งหมดเป็นของคุณ ทั้งสี ขนาด ไอคอน และการตัดข้อความ tag จะอยู่จนกว่าคุณจะเอาออกจาก state เอง

ติดตั้ง

ส่วน headless มากับแพ็กเกจเลย (ไม่มี item ของ shadcn; item ของ React + Tailwind สร้างบนมัน):

npm install @purpurabog/ui import { Tag } from '@purpurabog/ui/headless';

วิธีใช้

import { Tag } from '@purpurabog/ui/headless'; export default function Example() { return ( <Tag.Root className="inline-flex h-6 items-center rounded-full bg-tag-background-green px-2 text-xs text-tag-color-green"> ลงนามแล้ว </Tag.Root> ); }

โครงสร้าง

<Tag.Root> <Tag.Dismiss /> </Tag.Root> <Tag.Toggle />

แบบปิดได้

ใส่ Tag.Dismiss ไว้ใน Tag.Root แล้วลบ tag ใน onDismiss ตั้งชื่อปุ่มด้วย aria-label ให้ไม่ซ้ำกันในแต่ละ tag ถ้า tag ที่ลบมี focus อยู่ ให้ย้าย focus ไป tag ถัดไปเอง

import { useRef, useState } from 'react'; import { Tag } from '@purpurabog/ui/headless'; export default function Example() { const [filters, setFilters] = useState(['ฝ่ายขาย', 'ฝ่ายกฎหมาย', 'ลงนามปี 2569']); const row = useRef<HTMLDivElement>(null); const remove = (index: number) => { setFilters(filters.filter((_, i) => i !== index)); // หลัง React ลบออกแล้ว focus ปุ่มปิดที่อยู่ตำแหน่งเดิม (หรืออันสุดท้าย) requestAnimationFrame(() => { const buttons = row.current?.querySelectorAll('button'); buttons?.[Math.min(index, buttons.length - 1)]?.focus(); }); }; return ( <div className="grid gap-2"> <div ref={row} className="flex flex-wrap items-center gap-2"> {filters.map((filter, index) => ( <Tag.Root key={filter} className="inline-flex h-6 items-center rounded-full bg-tag-background-blue ps-2 text-xs text-tag-color-blue"> {filter} <Tag.Dismiss aria-label={`เอาตัวกรอง ${filter} ออก`} onDismiss={() => remove(index)} className="ms-0.5 flex size-6 items-center justify-center rounded-full hover:bg-tag-hover-blue focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-focus" > <svg viewBox="0 0 16 16" className="size-4 fill-current" aria-hidden="true"> <path d="m12 4.7-.7-.7L8 7.3 4.7 4l-.7.7L7.3 8 4 11.3l.7.7L8 8.7l3.3 3.3.7-.7L8.7 8z" /> </svg> </Tag.Dismiss> </Tag.Root> ))} </div> <p className="text-xs text-ink-muted">ตัวกรอง: {filters.join(', ') || 'ไม่มี'}</p> </div> ); }

แบบเปิดปิดได้

Tag.Toggle เก็บ state เองเมื่อใช้ defaultSelected หรือตาม selected เมื่อคุณคุมค่าเอง onSelectedChange ได้ค่าใหม่หลังการกดทุกครั้ง

import { useState } from 'react'; import { Tag } from '@purpurabog/ui/headless'; const departments = ['ฝ่ายขาย', 'ฝ่ายกฎหมาย', 'ฝ่ายการเงิน', 'ฝ่ายบุคคล']; export default function Example() { const [shown, setShown] = useState<string[]>(['ฝ่ายขาย']); return ( <div className="grid gap-2"> <div className="flex flex-wrap items-center gap-2"> {departments.map((dept) => ( <Tag.Toggle key={dept} selected={shown.includes(dept)} onSelectedChange={(on) => setShown(on ? [...shown, dept] : shown.filter((d) => d !== dept))} className="inline-flex h-6 items-center rounded-full border border-border-inverse bg-layer-1 px-2 text-xs text-ink hover:bg-layer-hover-1 focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-focus data-selected:bg-layer-selected-inverse data-selected:text-inverse-ink" > {dept} </Tag.Toggle> ))} </div> <p className="text-xs text-ink-muted">แสดง: {shown.join(', ') || 'ทุกแผนก'}</p> </div> ); }

แบบกดเพื่อเปิด

tag แบบกดเพื่อเปิดคือ <button type="button"> ของคุณเองที่วาดให้เป็น tag ถ้ามันแสดงหรือซ่อนอะไร ให้ใส่ aria-expanded (และ aria-controls) ถ้าข้อความเป็นแค่ตัวเลข ให้ตั้งชื่อด้วย aria-label

import { useId, useState } from 'react'; import { Tag } from '@purpurabog/ui/headless'; export default function Example() { const [open, setOpen] = useState(false); const listId = useId(); return ( <div className="grid gap-2"> <div className="flex flex-wrap items-center gap-2"> <Tag.Root className="inline-flex h-6 items-center rounded-full bg-tag-background-blue px-2 text-xs text-tag-color-blue">ฝ่ายขาย</Tag.Root> <Tag.Root className="inline-flex h-6 items-center rounded-full bg-tag-background-blue px-2 text-xs text-tag-color-blue">ฝ่ายกฎหมาย</Tag.Root> <button type="button" aria-label="อีก 3 แผนก" aria-expanded={open} aria-controls={listId} onClick={() => setOpen(!open)} className="inline-flex h-6 items-center rounded-full border border-tag-border-blue bg-tag-background-blue px-2 text-xs text-tag-color-blue hover:bg-tag-hover-blue focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-focus" > +3 </button> </div> <p id={listId} hidden={!open} className="text-sm text-ink"> ฝ่ายการเงิน, ฝ่ายจัดซื้อ, ฝ่ายบุคคล </p> </div> ); }

ปิดใช้งาน

disabled บน Tag.Root ปิด Tag.Dismiss ข้างในด้วย (ถ้า Dismiss มี disabled ของตัวเอง จะใช้ค่านั้น) Tag.Toggle รับ disabled เหมือนปุ่มทั่วไป แล้วแต่ละ part จะมี data-disabled

import { Tag } from '@purpurabog/ui/headless'; export default function Example() { return ( <div className="flex flex-wrap items-center gap-2"> <Tag.Root disabled className="inline-flex h-6 items-center rounded-full bg-tag-background-blue ps-2 text-xs text-tag-color-blue data-disabled:bg-layer-1 data-disabled:text-ink-disabled"> ฝ่ายกฎหมาย <Tag.Dismiss aria-label="เอาตัวกรอง ฝ่ายกฎหมาย ออก" className="ms-0.5 flex size-6 items-center justify-center rounded-full data-disabled:cursor-not-allowed"> <svg viewBox="0 0 16 16" className="size-4 fill-current" aria-hidden="true"> <path d="m12 4.7-.7-.7L8 7.3 4.7 4l-.7.7L7.3 8 4 11.3l.7.7L8 8.7l3.3 3.3.7-.7L8.7 8z" /> </svg> </Tag.Dismiss> </Tag.Root> <Tag.Toggle disabled defaultSelected className="inline-flex h-6 items-center rounded-full border border-border-inverse px-2 text-xs data-selected:bg-layer-selected-inverse data-selected:text-inverse-ink data-disabled:cursor-not-allowed data-disabled:border-ink-disabled data-disabled:bg-layer-1 data-disabled:text-ink-disabled"> ฝ่ายการเงิน </Tag.Toggle> </div> ); }

Parts

Tag.Root

<span> ตัว tag เอง รับ prop ของ <span> ได้ทุกตัว

Prop Type Default
disabled boolean false ปิด Tag.Dismiss ข้างในด้วย
Attribute เมื่อไร
data-disabled ตั้ง disabled

Tag.Dismiss

<button type="button"> ปุ่มปิด รับ prop ของปุ่มได้ทุกตัวยกเว้น type การกดจบที่ปุ่มนี้ click handler ที่ครอบ tag อยู่จะไม่เห็น

Prop Type Default
onDismiss () => void — หลังการกด ยกเว้นตอนปิดใช้งาน
disabled boolean ตาม root สถานะของตัวเอง ถ้าไม่ใส่จะตาม Tag.Root
Attribute เมื่อไร
data-disabled ปิดใช้งาน จากตัวเองหรือจาก Tag.Root

Tag.Toggle

<button type="button"> ที่มี aria-pressed คือ tag แบบเปิดปิดได้ รับ prop ของปุ่มได้ทุกตัวยกเว้น type และ aria-pressed

Prop Type Default
selected boolean — สถานะ (controlled)
defaultSelected boolean false สถานะเริ่มต้น (uncontrolled)
onSelectedChange (selected: boolean, event: MouseEvent<HTMLButtonElement>) => void — หลังผู้ใช้สลับ
Attribute เมื่อไร
data-selected เปิดอยู่
data-disabled ตั้ง disabled

คีย์บอร์ดและ accessibility

ปุ่ม ทำอะไร
Tab ไปที่ปุ่มปิดหรือ tag แบบเปิดปิดได้ ตัว Tag.Root ไม่ใช่จุดที่ Tab หยุด
Enter, Space กดปุ่มปิด หรือสลับ tag แบบเปิดปิดได้

part ตั้ง type="button", aria-pressed บน Tag.Toggle และ disabled แบบ native ให้

สิ่งที่คุณต้องเพิ่มเอง:

  • ชื่อของ Tag.Dismiss แต่ละอัน เช่น aria-label="เอาตัวกรอง ฝ่ายกฎหมาย ออก"
  • focus หลังการลบ เพราะปุ่มที่ถูกลบพา focus หายไปด้วย
  • aria-label บน tag แบบกดเพื่อเปิดที่มีแค่ตัวเลข และ aria-expanded เมื่อมันแสดงหรือซ่อนอะไร
  • ข้อความเต็มใน title เมื่อตัด tag ยาวด้วย ...
  • style ของ focus ที่มองเห็นได้
